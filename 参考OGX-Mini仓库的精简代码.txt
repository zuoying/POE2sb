模块 3：WS2812 LED 状态指示（3 模式 + 低亮度）
源文件：src/led/ws2812.c（精简版）
// OGX-Mini WS2812驱动 → 你的3模式指示灯
#include "hardware/pio.h"
#include "ws2812.pio.h"

#define LED_PIN 16
#define LED_BRIGHTNESS 32    // 正常亮度
#define LED_LOW_BRIGHTNESS 8 // 无手柄时低亮度

// 初始化LED
void ws2812_init(void) {
  PIO pio = pio0;
  uint sm = pio_claim_unused_sm(pio, true);
  uint offset = pio_add_program(pio, &ws2812_program);
  ws2812_program_init(pio, sm, offset, LED_PIN, 800000, false);
}

// 设置LED颜色+亮度
void ws2812_set_color(uint8_t r, uint8_t g, uint8_t b, uint8_t brightness) {
  uint32_t color = (r * brightness / 255) << 16 |
                   (g * brightness / 255) << 8  |
                   (b * brightness / 255);
  pio_sm_put_blocking(pio0, 0, color << 8u);
}

// 模式LED定义
#define LED_SYNC_MODE    0,255,0  // 绿灯 同步模式
#define LED_MAIN_MODE    0,0,255  // 蓝灯 仅主角色
#define LED_SUB_MODE     255,0,0  // 红灯 仅副角色

// OGX-Mini 模式管理 → 你的3模式切换框架
#include "pico/stdlib.h"

// 3种工作模式
typedef enum {
  SYNC_MODE,    // 同步
  MAIN_MODE,    // 仅主
  SUB_MODE      // 仅副
} work_mode_t;

work_mode_t current_mode = SYNC_MODE;

// 组合键切换模式（示例：Menu+View长按切换）
void check_mode_switch(bool menu_pressed, bool view_pressed) {
  static uint32_t press_time = 0;
  if (menu_pressed && view_pressed) {
    press_time += 10;
    if (press_time >= 1000) { // 长按1秒切换
      current_mode = (current_mode + 1) % 3;
      press_time = 0;
    }
  } else {
    press_time = 0;
  }
}
